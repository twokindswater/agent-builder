# AI Agent Builder - Backend 상세 설계서 (ADK Agent 기반)

## 1. 개요
이 문서는 Google ADK (Agent Development Kit)를 활용하여 AI Agent Builder의 백엔드를 구현하는 상세 설계서입니다. ADK의 LLM Agent, MCP Tool, Runner 기능을 활용하여 프론트엔드와 통신하는 RESTful API를 제공합니다.

---

## 2. 기술 스택 및 아키텍처

### 2.1. 기술 스택
- **Framework**: FastAPI (Python 3.12+)
- **AI Agent Framework**: Google ADK (Agent Development Kit)
- **Database**: Supabase (PostgreSQL)
- **Authentication**: Supabase Auth (JWT)
- **Task Queue**: Celery + Redis (비동기 워크플로우 실행)
- **Real-time Communication**: WebSocket (FastAPI-WebSocket)
- **ORM**: SQLAlchemy with Supabase
- **Environment Management**: Python-dotenv

### 2.2. ADK 컴포넌트 활용
- **LLM Agent**: OpenAI 모델을 활용한 AI Agent 생성 및 관리
- **MCP Tools**: MCP 서버의 도구들을 Agent에 연결
- **Runner**: 생성된 LLM Agent 실행 및 세션 관리
- **Session Service**: Agent 실행 컨텍스트 관리

### 3.1. 테이블 설계
```sql
-- 사용자 테이블 (Supabase Auth 기본 제공)
-- users: id, email, password, created_at, updated_at

-- AI Agent 테이블
CREATE TABLE agents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    system_prompt TEXT NOT NULL,
    model_config JSONB NOT NULL DEFAULT '{
        "model": "gpt-4",
        "temperature": 0.7,
        "max_tokens": 1000
    }',
    template_type VARCHAR(100), -- 'chatbot', 'data_analyst', 'code_generator', etc.
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- MCP 서버 테이블
CREATE TABLE mcp_servers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    url VARCHAR(500) NOT NULL,
    encrypted_token TEXT, -- 암호화된 인증 토큰
    description TEXT,
    is_connected BOOLEAN DEFAULT false,
    last_connected_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- MCP 도구 테이블 (서버에서 제공하는 도구들)
CREATE TABLE mcp_tools (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    server_id UUID REFERENCES mcp_servers(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    input_schema JSONB,
    output_schema JSONB,
    is_available BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 워크플로우 테이블
CREATE TABLE workflows (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    definition JSONB NOT NULL, -- 워크플로우 노드 및 연결 정보
    version INTEGER DEFAULT 1,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 워크플로우 실행 기록 테이블
CREATE TABLE workflow_executions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workflow_id UUID REFERENCES workflows(id) ON DELETE CASCADE,
    session_id VARCHAR(255) NOT NULL, -- ADK Session ID
    status VARCHAR(50) NOT NULL DEFAULT 'pending', -- 'pending', 'running', 'completed', 'failed'
    input_data JSONB,
    output_data JSONB,
    execution_logs JSONB DEFAULT '[]',
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    finished_at TIMESTAMP WITH TIME ZONE,
    error_message TEXT
);

-- Agent와 MCP Tool 연결 테이블 (Many-to-Many)
CREATE TABLE agent_mcp_tools (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id UUID REFERENCES agents(id) ON DELETE CASCADE,
    tool_id UUID REFERENCES mcp_tools(id) ON DELETE CASCADE,
    is_enabled BOOLEAN DEFAULT true,
    tool_config JSONB DEFAULT '{}', -- 도구별 설정
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(agent_id, tool_id)
);
```

        agent = LlmAgent(
            model=agent_data['model_config']['model'],
            name=agent_data['name'],
            description=agent_data['description'],
            instruction=agent_data['system_prompt'],

        # MCP 서버 API 호출하여 실제 도구 함수 생성
        # ADK Tool 형태로 변환하여 반환
        pass
    
    async def execute_agent(self, agent_id: str, input_data: dict) -> dict:
        """Agent 실행"""
        runner = self.active_runners.get(agent_id)
        if not runner:
            raise ValueError(f"Agent {agent_id} not found or not initialized")

import httpx
from typing import List, Dict, Any

class MCPService:
    def __init__(self):
        self.connected_servers = {}  # server_id -> connection info
    
    async def connect_server(self, server_data: dict) -> bool:
        """MCP 서버 연결 및 도구 목록 조회"""
        try:
            async with httpx.AsyncClient() as client:
                # MCP 서버 연결 테스트
                response = await client.get(
                    f"{server_data['url']}/tools",
                    headers={"Authorization": f"Bearer {server_data['token']}"}
                )
                
                if response.status_code == 200:
                    tools = response.json()
                    await self._sync_tools(server_data['id'], tools)
                    self.connected_servers[server_data['id']] = server_data
                    return True
                    
        except Exception as e:
            print(f"MCP 서버 연결 실패: {e}")
            return False
    
    async def _sync_tools(self, server_id: str, tools: List[Dict]) -> None:
        """MCP 서버의 도구 목록을 데이터베이스에 동기화"""
        # 기존 도구 목록 삭제
        # 새로운 도구 목록 저장
        pass
    
    def create_tool_function(self, tool_data: dict) -> callable:
        """MCP 도구 정보로부터 ADK에서 사용할 수 있는 함수 생성"""
        async def tool_function(**kwargs) -> str:
            server_id = tool_data['server_id']
            server_info = self.connected_servers.get(server_id)
            
            if not server_info:

                )
                
                return response.json()
        
        # 함수 메타데이터 설정
        tool_function.__name__ = tool_data['name']
        tool_function.__doc__ = tool_data['description']

## 7. 보안 및 인증

### 7.1. JWT 토큰 검증
```python
# middleware/auth.py
from fastapi import HTTPException, Depends
from fastapi.security import HTTPBearer
import jwt

security = HTTPBearer()

async def verify_token(token: str = Depends(security)):
    """Supabase JWT 토큰 검증"""
    try:
        # Supabase JWT 검증 로직
        payload = jwt.decode(token.credentials, options={"verify_signature": False})
        return payload
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="유효하지 않은 토큰")
```

### 7.2. MCP 서버 토큰 암호화
```python
# utils/encryption.py
from cryptography.fernet import Fernet
import os

class TokenEncryption:
    def __init__(self):
        self.key = os.getenv('ENCRYPTION_KEY', Fernet.generate_key())
        self.fernet = Fernet(self.key)
    
    def encrypt_token(self, token: str) -> str:
        """토큰 암호화"""
        return self.fernet.encrypt(token.encode()).decode()
    
    def decrypt_token(self, encrypted_token: str) -> str:
        """토큰 복호화"""
        return self.fernet.decrypt(encrypted_token.encode()).decode()
```
